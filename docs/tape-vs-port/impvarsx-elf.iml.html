1#IMPVARSX-ELF: THE SECTION impvars OF AN ELF PROGRAM, WHICH THE LINKER BOUNDS WITH __START_IMPVARS AND __STOP_IMPVARS, AND THE WRITABLE PART OF THE PROGRAM: ITS PT_LOAD SEGMENTS WITH PF_W, FROM DL_ITERATE_PHDR (THE
2FIRST OBJECT IT SHOWS IS THE PROGRAM). THE LAYOUT OF A PROGRAM HEADER IS THAT OF THE ELF OF THE SIZE OF A POINTER (32 OR 64 BITS).#
4__start_impvars,__stop_impvars ARE EXTERNAL;
5WRL IS 16 LONG;
6WRH IS 16 LONG;
8SUBR VSTART(Z) IS LOC(__start_impvars);
10SUBR VSTOP(Z) IS LOC(__stop_impvars);
12# THE CALLBACK OF DL_ITERATE_PHDR(INFO,SIZE,DATA): ITS PARAMETERS ARE THE C LIBRARY'S POINTERS, WHICH THE ROUTINE TAKES FOR POINTERS TO WORDS, SO THAT LOC OF ONE IS THE POINTER. THE FIELDS OF A DL_PHDR_INFO ARE DLPI_ADDR,
13 DLPI_NAME, DLPI_PHDR AND (A HALF WORD) DLPI_PHNUM #
14SUBR VPHDR(INFO,SIZE,DATA) IS (
15 P,I,N,PH,BASE,PS,Q,FL,VA,MS ARE LOCAL;
16 P:=LOC(INFO);
17 BASE:=IMPPEEKPTR(P); PH:=IMPPEEKPTR(P+(2*SCPTR(0))); N:=IMPPEEK(P+(3*SCPTR(0)),2);
18 PS:=(SCPTR(0)=8 => 56 ELSE 32);
19 I:=0;
20 VP1: I<N => (
21 Q:=PH+(I*PS);
22 IMPPEEK(Q,4)=1 => ( # PT_LOAD #
23 SCPTR(0)=8 => (FL:=IMPPEEK(Q+4,4); VA:=IMPPEEK(Q+16,8); MS:=IMPPEEK(Q+40,8))
24 ELSE (FL:=IMPPEEK(Q+24,4); VA:=IMPPEEK(Q+8,4); MS:=IMPPEEK(Q+20,4));
25 (FL AND 2) NE 0 => (NWR<16 => (WRL[NWR]:=BASE+VA; WRH[NWR]:=(BASE+VA)+MS; NWR:=NWR+1)));
26 I:=I+1; GO TO VP1);
27 1);
29# IS THE VARIABLE AT A, S BYTES LONG, IN THE WRITABLE PART OF THE PROGRAM #
30SUBR WRITABLE(A,S) IS (
31 I ARE LOCAL;
32 NWR=0 => CCALL dl_iterate_phdr(LOC(VPHDR),0);
33 I:=0;
34 WR1: I<NWR => (A GE WRL[I] => ((A+S) LE WRH[I] => RETURN 1);
35 I:=I+1; GO TO WR1);
36 0)
