14# CALLI(CODE,AC): THE TOPS-10 MONITOR CALLS THE COMPILER'S SOURCES MAKE. 0 RESET, 012 EXIT, 014 THE TIME AS UNIX TIME (SECONDS SINCE 1970-01-01 00:00:00 UTC; THE MONITOR'S WAS THE DATE AS ((YEAR-1964)*12+MONTH-1)*31+DAY-1), 023 THE SAME IN MILLISECONDS (TO THE SECOND;

16SUBR impcalli(CODE,AC) IS (
17 CODE=10 => (CCALL fflush(IMPSTDOUT(0)); CCALL exit(0));

19 CODE=19 => RETURN IMPTIME(0)*1000;
20 CODE=23 => RETURN CCALL clock(0)/1000;
21 CODE=24 => RETURN CCALL getpid(0) AND 4294967295;
22 0);
24# BLT: COPY BYTES BYTES FROM SRC TO DST, A WORD AT A TIME UPWARD AS THE PDP-10 DOES: WHEN THE RANGES OVERLAP WITH DST ABOVE SRC THE WORDS COPIED ARE THE WORDS ALREADY COPIED (A WORD STORED AND BLT'D THROUGH AN
25 ARRAY FILLS IT; MEMMOVE WOULD NOT) #
26SUBR impblt(DST,SRC,BYTES) IS (
27 I ARE LOCAL;
28 I:=0;
29 BL1: I GE BYTES => RETURN 0;

32# IFIX(X): A REAL TO AN INTEGER, TRUNCATING (THE LIBRARY ROUTINE THE ORIGINAL REACHES AS MALAMUD). THE ARGUMENT IS THE ADDRESS OF THE REAL, WHICH IS THE BITS OF AN IEEE DOUBLE. A VALUE TOO BIG IS THE
33 LEAST INTEGER, AS THE MACHINE GIVES IT; THE ROUTINE CANNOT USE FIX #
34SUBR IFIX(X) IS (
35 B,E,M,NG,U,V ARE LOCAL;
36 B:=X; NG:=(B<0 => 1 ELSE 0);
37 E:=(B RS 52) AND 2047;
38 E<1023 => RETURN 0; # LESS THAN 1 #
39 U:=E-1023;
40 U>62 => RETURN 0-9223372036854775807-1;
41 M:=(B AND 4503599627370495) OR 4503599627370496;
42 U<52 => V:=M RS (52-U) ELSE V:=M LS (U-52);
43 NG NE 0 => 0-V ELSE V);
45# THE SHIFTS AND THE ROTATION OF A WORD BY A VARIABLE COUNT (THE CODE GENERATOR IN SRC CALLS THEM): A POSITIVE COUNT IS TO THE LEFT, A NEGATIVE ONE TO THE RIGHT; A LOGICAL SHIFT OF 64 PLACES OR MORE GIVES 0. THE
46 ARGUMENTS AND THE RESULT ARE THE ADDRESSES OF 64 BIT WORDS #
47SUBR implsh(PX,PN,PR) IS (
48 X,N,NEG64 ARE LOCAL;
49 X:=PX; N:=PN; NEG64:=0-64;
50 N GE 64 => (PR:=0; RETURN 0);
51 N LE NEG64 => (PR:=0; RETURN 0);
52 N GE 0 => PR:=X LS N ELSE PR:=X RS (0-N);
53 0);

58 N GE 64 => (PR:=0; RETURN 0);
59 N GE 0 => (PR:=X LS N; RETURN 0);
60 N LE NEG64 => N:=0-63;

71# BYTE POINTERS. A POINTER TO CHARACTERS (THE TAPE'S BYTEP A<7,36>) IS A BYTE ADDRESS: IBP ADDS 1, LDB READS THE BYTE THERE, DPB WRITES THE LOW 8 BITS OF THE VALUE TO IT (A CHARACTER IS A BYTE, UTF-8). ANY OTHER POINTER
72 NAMES A BIT FIELD AND KEEPS THE LAYOUT OF THE PDP-10'S, IN ONE WORD: BITS 63-58 THE SIZE, 57-52 THE POSITION (THE LOW BIT OF THE FIELD, COUNTED FROM THE RIGHT), THE REST THE BYTE ADDRESS OF THE WORD; IBP MOVES TO THE NEXT
73 LOWER FIELD OF THE SAME SIZE, OR TO THE NEXT WORD. A BYTE ADDRESS IS BELOW 2**47, SO THE SIZE BITS TELL THE TWO APART. THE POINTER IS IN IMP_BPP AND THE BYTE IN IMP_BPV #
74SUBR impibp(Z) IS (
75 P,SZ,POS,NP ARE LOCAL;
76 P:=impbpp; (P RS 32)=4294967295 => P:=P AND 4294967295; # A 32 BIT ADDRESS THAT A CCALL sign EXTENDED IS THE ADDRESS, NOT A FIELD #
77 SZ:=P RS 58; POS:=(P RS 52) AND 63;
78 SZ=0 => (impbpp:=P+1; RETURN 0);
79 NP:=POS-SZ;
80 NP<0 => (NP:=64-SZ; P:=P+8);
81 impbpp:=(P AND (NOT (63 LS 52))) OR (NP LS 52);
82 0);

113# EXECUTE: IMP_XCT() DOES THE ONE INSTRUCTION OF THE IR MACHINE (DOCS/PORTING/LLVM-IR-MACHINE.MD) THAT THE WORD IMP_XW HOLDS, THE WAY THE COMPILED PROGRAM WOULD HAVE DONE IT WHERE IT STANDS. AN INSTRUCTION WORD HAS THE

115 47 SET FOR AN IMMEDIATE OPERAND, AND IN BITS 0-46 THE ADDRESS OR THE NUMBER (SIGNED). THE ACCUMULATORS ARE THE WORDS OF IMP_AC, WHICH THE COMPILED CODE FILLS FROM ITS REGISTERS 0R TO 15R BEFORE THE CALL AND READS BACK
116 AFTER. AS IN COMPILED CODE AN ADDRESS IS A BYTE ADDRESS: WITH AN INDEX REGISTER IT IS THAT REGISTER PLUS 8 TIMES THE ADDRESS FIELD (A WORD INDEX), WITHOUT ONE IT IS THE FIELD ITSELF; AN IMMEDIATE OPERAND IS THE FIELD PLUS
